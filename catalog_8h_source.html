3#define NEW_UNIT_LITERAL(QuantityType, short_name, long_name, UnitType, ScaleNum, ScaleDen) \
4 struct UnitType { \
5 typedef typename dimension_for_quantity<QuantityType>::type dimension; \
6 static constexpr double scale_num = static_cast<double>(ScaleNum); \
7 static constexpr double scale_den = static_cast<double>(ScaleDen); \
8 static constexpr bool is_affine = false; \
9 static const char* symbol() { return #short_name; } \
10 static const char* name() { return #long_name; } \
11 }; \
12 namespace literals { \
13 constexpr QuantityType operator""_##short_name(long double value) { \
14 return QuantityType::from<UnitType>(static_cast<double>(value)); \
15 } \
16 constexpr QuantityType operator""_##short_name(unsigned long long value) { \
17 return QuantityType::from<UnitType>(static_cast<double>(value)); \
18 } \
